# scripts/visual/city_creation_wall_material.gd
extends RefCounted
const WallShader := preload("res://scripts/visual/shaders/creation_wall.gdshader")
enum Kind { PLASTER, STONE, TERRACOTTA, SLATE, CANTILEVER, TIMBER, METAL }

static func source(color: Color, texture: Texture2D = null, style: String = "classic", kind: int = Kind.PLASTER) -> StandardMaterial3D:
	var material := StandardMaterial3D.new()
	material.albedo_color = color
	material.albedo_texture = texture
	material.texture_filter = BaseMaterial3D.TEXTURE_FILTER_LINEAR_WITH_MIPMAPS
	material.texture_repeat = true
	if style == "stone" and texture != null:
		material.uv1_scale = Vector3(0.68, 0.68, 0.68)
	material.roughness = 0.96 if style == "stone" else (0.62 if kind == Kind.TIMBER else (0.38 if kind == Kind.METAL else 0.90))
	material.cull_mode = BaseMaterial3D.CULL_DISABLED if style == "stone" else BaseMaterial3D.CULL_BACK
	material.set_meta("creation_wall", true)
	material.set_meta("creation_wall_style", style)
	material.set_meta("creation_surface_kind", kind)
	return material

补丁 3：内凹窗洞的吹制玻璃材质生成函数（scripts/visual/city_creation_renderer.gd）
对应 P2 阶段，在 city_creation_renderer.gd 中添加 _window_glass_material(seed_value: int) 辅助方法（供上文 3.1 节改造后的 _facade() 调用）。它标记了 creation_wall 以维持 MeshBatcher 合批，并通过低粗糙度与微调色差赋予日间窗户清透的蓝绿天光反射感：
code
Gdscript
# 添加至 scripts/visual/city_creation_renderer.gd (_material 函数下方)
func _window_glass_material(seed_value: int) -> StandardMaterial3D:
	var variant := posmod(seed_value, 4)
	var key := "window_glass_%d" % variant
	if materials.has(key):
		return materials[key]
	var tints := [
		Color("355256"),
		Color("2f4b52"),
		Color("3b5759"),
		Color("324d4e")
	]
	var mat := WallMaterial.source(tints[variant], null, "component", WallMaterial.Kind.SLATE)
	mat.roughness = 0.22
	materials[key] = mat
	return mat
