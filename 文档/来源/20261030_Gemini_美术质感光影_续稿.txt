        GrassMask 禁草边界完全不变，仅用带中央叶脉折脊的双坡面替代冗余封闭圆管：
    # scripts/visual/city_creation_meadow_fold.gd :: _rounded() 替换为轻量立体折脊叶
    static func _rounded(s: SurfaceTool, root: Vector3, outward: Vector3, across: Vector3, height: float, reach: float, width: float, index: int) -> void:
        var steps := 4
        var t_vals := [0.0, 0.28, 0.58, 0.84, 1.0]
        var w_vals := [0.42, 0.95, 1.0, 0.58, 0.0]
        var ridge_lift := 0.22 # 叶脉中央隆起比例，形成左右受光分明的 V 型折面
        for i in range(steps):
            var t0: float = t_vals[i]
            var t1: float = t_vals[i + 1]
            var c0 := _center(root, outward, height, reach, t0, index)
            var c1 := _center(root, outward, height, reach, t1, index)
            var tangent := (c1 - c0).normalized()
            var up_axis := across.cross(tangent).normalized()
            var w0: float = width * w_vals[i]
            var w1: float = width * w_vals[i + 1]
            var r0 := c0 + up_axis * (w0 * ridge_lift)
            var r1 := c1 + up_axis * (w1 * ridge_lift)
            var l0 := c0 - across * w0
            var l1 := c1 - across * w1
            var rt0 := c0 + across * w0
            var rt1 := c1 + across * w1
            var n_left := (up_axis * 0.78 - across * 0.62).normalized()
            var n_right := (up_axis * 0.78 + across * 0.62).normalized()
            # 左半叶面与右半叶面沿着中央叶脉 r0-r1 折合，斜阳下形成清晰的立体明暗脊线
            for tri: Array in [[l0, r0, r1, n_left], [l0, r1, l1, n_left], [r0, rt0, rt1, n_right], [r0, rt1, r1, n_right]]:
                var n: Vector3 = tri[3]
                for k in range(3):
                    var p: Vector3 = tri[k]
                    s.set_normal(n)
                    s.set_color(Color.WHITE)
                    s.set_uv(Vector2(float(index + 1), clampf(p.y / height, 0.0, 1.0)))
                    s.add_vertex(p)
