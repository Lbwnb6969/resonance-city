// FLOAT/INT reinterpret bits; BITS stores float bits; BITS4 raw vector.

    uint4 r0;
    uint4 r1;
    uint4 out_o0;
    uint4 out_o1;
    uint4 v0;
    uint4 v1;

    v1.xy = asuint(input.v1_0);
