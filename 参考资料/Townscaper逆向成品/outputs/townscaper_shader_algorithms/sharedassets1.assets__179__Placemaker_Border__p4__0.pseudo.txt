// FLOAT/INT reinterpret bits; BITS stores float bits; BITS4 raw vector.

    uint4 r0;
    uint4 r1;
    uint4 out_o0;
    uint4 out_o1;
    uint4 out_o2;
    uint4 v0;
    uint4 v1;
    v0.xyzw = asuint(input.v0_0);
    v1.xy = asuint(input.v1_0);
    r0.xyz = (BITS(FLOAT(v0.yyyy) * FLOAT(cb0[1].xyzx))).xyz; // mul r0.xyz, v0.yyyy, cb0[1].xyzx
    r0.xyz = (BITS(mad(FLOAT(cb0[0].xyzx), FLOAT(v0.xxxx), FLOAT(r0.xyzx)))).xyz; // mad r0.xyz, cb0[0].xyzx, v0.xxxx, r0.xyzx
    r0.xyz = (BITS(mad(FLOAT(cb0[2].xyzx), FLOAT(v0.zzzz), FLOAT(r0.xyzx)))).xyz; // mad r0.xyz, cb0[2].xyzx, v0.zzzz, r0.xyzx
    r0.xyz = (BITS(mad(FLOAT(cb0[3].xyzx), FLOAT(v0.wwww), FLOAT(r0.xyzx)))).xyz; // mad r0.xyz, cb0[3].xyzx, v0.wwww, r0.xyzx
    r1.xyzw = (BITS(FLOAT(r0.yyyy) * FLOAT(cb1[18].xyzw))).xyzw; // mul r1.xyzw, r0.yyyy, cb1[18].xyzw
    r1.xyzw = (BITS(mad(FLOAT(cb1[17].xyzw), FLOAT(r0.xxxx), FLOAT(r1.xyzw)))).xyzw; // mad r1.xyzw, cb1[17].xyzw, r0.xxxx, r1.xyzw
    r0.xyzw = (BITS(mad(FLOAT(cb1[19].xyzw), FLOAT(r0.zzzz), FLOAT(r1.xyzw)))).xyzw; // mad r0.xyzw, cb1[19].xyzw, r0.zzzz, r1.xyzw

    out_o1.x = (BITS(frac(FLOAT(v1.xxxx)))).x; // frc o1.x, v1.x
    out_o1.y = (v1.yyyy).y; // mov o1.y, v1.y
    out_o2.w = (BITS(floor(FLOAT(v1.xxxx)))).w; // round_ni o2.w, v1.x
    out_o2.xyz = (BITS4(0x3f800000u, 0x3f800000u, 0x3f800000u, 0x0u)).xyz; // mov o2.xyz, l(1.000000,1.000000,1.000000,0)
    // ret: return all declared output registers below.
    Output result;
    result.o0_0 = asfloat(out_o0.xyzw);
    result.o1_0 = asfloat(out_o1.xy);
    result.o2_0 = asfloat(out_o2.xyzw);
