var failures: Array[String] = []

func _initialize() -> void: call_deferred("_run")

func check(ok: bool,label: String) -> void:
	checks[label] = ok
	if not ok:
		failures.append(label)
		push_error(label)

func _area_outside(polygon: PackedVector2Array,masks: Array[PackedVector2Array]) -> float:
	var remains: Array[PackedVector2Array] = [polygon]
	for mask: PackedVector2Array in masks:
		var next: Array[PackedVector2Array] = []
		for part: PackedVector2Array in remains:
			if not Spatial.bounds(part).intersects(Spatial.bounds(mask),true): next.append(part)
			else: next.append_array(Geometry2D.clip_polygons(part,mask))
		remains = next
		if remains.is_empty(): break
	var area := 0.0
	for part: PackedVector2Array in remains: area += Spatial.polygon_area(part)
	return area

func _overlap_area(a: PackedVector2Array,b: PackedVector2Array) -> float:
	if not Spatial.bounds(a).intersects(Spatial.bounds(b),true): return 0.0
	var area := 0.0
	for part: PackedVector2Array in Geometry2D.intersect_polygons(a,b): area += Spatial.polygon_area(part)
	return area

func _terminal_seed(city: RefCounted) -> Dictionary:
	var degree: Dictionary = {}
	for road: Dictionary in city.roads:
		degree[str(road["from"])] = int(degree.get(str(road["from"]),0))+1
		degree[str(road["to"])] = int(degree.get(str(road["to"]),0))+1
	for road: Dictionary in city.roads:
		if bool(road.get("bridge",false)): continue
		var path: PackedVector2Array = road["path"]
		for end: int in [0,1]:
			var node := str(road["from"] if end == 0 else road["to"])
			if int(degree[node]) != 1: continue
			var p := path[0] if end == 0 else path[-1]
			var into := path[0].direction_to(path[1]) if end == 0 else path[-1].direction_to(path[-2])
			var normal := Vector2(-into.y,into.x)
			for side: float in [-1.0,1.0]:
				var candidate := p+normal*(Miniature.road_width(city,road)+0.65)*side
				if bool(city.placement(candidate).get("allowed",false)):
					return {"position":candidate,"node":node,"road_id":str(road["id"])}
	return {}

func _run() -> void:

	var all_roads: Array[PackedVector2Array] = []
	var masks_by_id: Dictionary = {}
	var roads_by_id: Dictionary = {}
	for road: Dictionary in city.roads:
		var masks: Array[PackedVector2Array] = []
		masks.assign(Geometry2D.offset_polyline(road["path"],Miniature.road_width(city,road),Geometry2D.JOIN_ROUND,Geometry2D.END_BUTT))
		all_roads.append_array(masks)
		masks_by_id[str(road["id"])] = masks
		roads_by_id[str(road["id"])] = road
	var bounded := true
	var no_water_or_houses := true
	var max_outside := 0.0
	var max_overlap := 0.0
	var height_min := INF
	var height_max := -INF
	for kind: String in ["paving","stones","curbs"]:
		for piece: Dictionary in original[kind]:
			var polygon: PackedVector2Array = piece["polygon"]
			var outside := _area_outside(polygon,all_roads)
			max_outside = maxf(max_outside,outside)
			bounded = bounded and outside < 0.00001 and Spatial.polygon_area(polygon) > 0.000001 and not Geometry2D.triangulate_polygon(polygon).is_empty()
			var overlap := _overlap_area(polygon,city.terrain.water)
			for building: Dictionary in city.buildings: overlap = maxf(overlap,_overlap_area(polygon,city.building_polygon(Coupling.position_of(building))))
			for lot: Dictionary in city.lots:
				if not str(lot["occupant"]).is_empty(): overlap = maxf(overlap,_overlap_area(polygon,lot["polygon"]))
			for group: Array in [city.public_sites,city.planning_reservations]:
				for record: Dictionary in group: overlap = maxf(overlap,Spatial.footprint_intersection_area(polygon,record))
			max_overlap = maxf(max_overlap,overlap)
			no_water_or_houses = no_water_or_houses and overlap < 0.00001
			for vertex: Vector2 in polygon:
				height_min = minf(height_min,city.terrain.height_at(vertex))
				height_max = maxf(height_max,city.terrain.height_at(vertex))
	check(bounded and is_equal_approx(Miniature.road_width(city,roads_by_id["west-road-2"]),0.25) and is_equal_approx(Miniature.road_width(city,roads_by_id["west-cross-2"]),0.18),"all_surface_areas_stay_in_actual_025_main_and_018_branch_road_union")
	check(no_water_or_houses,"stone_and_curb_areas_do_not_cover_water_real_houses_occupied_lots_or_owned_land")
	var crossing_clear := true
	var fixed_edges := true
	var branches := false
	var curved := false
	for paving: Dictionary in original["paving"]:
		var road: Dictionary = roads_by_id[str(paving["road_id"])]
		branches = branches or str(road.get("kind","")) == "branch"
		curved = curved or (road["path"].size() > 2 and paving["polygon"].size() > 4)
	for curb: Dictionary in original["curbs"]:
		var edge: PackedVector2Array = curb["edge"]
		fixed_edges = fixed_edges and edge.size() == 2 and float(curb["raise"]) > 0.0 and float(curb["raise"]) <= 0.018 and _length(edge) > 0.005
		for other_id: String in masks_by_id:
			if other_id == str(curb["road_id"]): continue
			for mask: PackedVector2Array in masks_by_id[other_id]:
				crossing_clear = crossing_clear and _overlap_area(curb["polygon"],mask) < 0.00001
	check(crossing_clear and fixed_edges and branches and curved,"real_curved_main_branch_junctions_remain_open_with_whole_curb_area_clipped_and_two_vertex_edges")

	check(float(coverage["expected_length"]) > 10.0 and float(coverage["minimum_road_length_ratio"]) > 0.90 and float(coverage["maximum_road_length_ratio"]) < 1.08 and float(coverage["sample_coverage"]) > 0.97,"low_curbs_cover_actual_paved_longitudinal_boundaries_instead_of_only_a_few_coincident_fragments")
	var tones: Dictionary = {}
	var lengths: Dictionary = {}
	var rows: Dictionary = {}
	var tones_bounded := true
	for stone: Dictionary in original["stones"]:
		tones[stone["tone"]] = true
		lengths[snappedf(float(stone["length"]),0.001)] = true
		rows[stone["row"]] = true
		tones_bounded = tones_bounded and absf(float(stone["tone"])) <= 0.035001

	check(tones_bounded and tones.size() > 8 and lengths.size() > 8 and rows.size() == 3 and original == repeated.material and original["joints"].is_empty(),"unequal_staggered_stones_use_stable_local_tones_and_real_mortar_gaps_without_world_rng")
	var retained_key: Array = surface.key
	var retained_stones: Array = surface.material["stones"]
	var rebuilds := surface.rebuild_count
	surface.sync(city,127)
	surface.sync(city,799)
	var vertex: Vector2 = original["stones"][0]["polygon"][0]
	var a := IsoProjection.to_screen(vertex-Vector2(16,16),Vector2.ZERO,0.7,Vector2(28,14),city.terrain.height_at(vertex))

static func _length(path: PackedVector2Array) -> float:
	var result := 0.0
	for i in range(path.size()-1): result += path[i].distance_to(path[i+1])
	return result

func _curb_coverage(material: Dictionary,city: RefCounted,roads: Dictionary,masks: Dictionary) -> Dictionary:
	var expected: Dictionary = {}
	var actual: Dictionary = {}
	var samples := 0
	var covered := 0
	var side_edges: Array[PackedVector2Array] = []
	for curb: Dictionary in material["curbs"]:
		if curb["kind"] != "side": continue
		var id := str(curb["road_id"])
		actual[id] = float(actual.get(id,0.0))+_length(curb["edge"])
		side_edges.append(curb["edge"])
	# Independently find the real longitudinal edges of the delivered bed.
	# Cross streets are subtracted in full; occupancy cuts and end caps are not
	# longitudinal edges and therefore do not inflate the expected coverage.
	for paving: Dictionary in material["paving"]:
		var id := str(paving["road_id"])
		var road: Dictionary = roads[id]
		var width := Miniature.road_width(city,road)
		var path: PackedVector2Array = road["path"]
		var polygon: PackedVector2Array = paving["polygon"]
		for i in range(polygon.size()):
			var a := polygon[i]
			var b := polygon[(i+1)%polygon.size()]
			var midpoint := (a+b)*0.5
			var nearest := INF
			for j in range(path.size()-1): nearest = minf(nearest,midpoint.distance_to(Geometry2D.get_closest_point_to_segment(midpoint,path[j],path[j+1])))
			if absf(nearest-width) > 0.009 or a.distance_to(b) < 0.005: continue
			var parts: Array[PackedVector2Array] = [PackedVector2Array([a,b])]
			for other_id: String in masks:
				if other_id == id: continue
				for mask: PackedVector2Array in masks[other_id]:
					var clipped: Array[PackedVector2Array] = []
					for part: PackedVector2Array in parts: clipped.append_array(Geometry2D.clip_polyline_with_polygon(part,mask))
					parts = clipped
			for part: PackedVector2Array in parts:
				expected[id] = float(expected.get(id,0.0))+_length(part)
				var count := maxi(1,ceili(_length(part)/0.08))
				for step in range(count):
					var sample := part[0].lerp(part[-1],(float(step)+0.5)/float(count))
					samples += 1
					var matched := false
					for edge: PackedVector2Array in side_edges:
						if sample.distance_to(Geometry2D.get_closest_point_to_segment(sample,edge[0],edge[1])) < 0.013:
							matched = true
							break
					if matched: covered += 1
	var minimum := INF
	var maximum := 0.0
	var total := 0.0
	for id: String in expected:
		if float(expected[id]) < 0.05: continue
		var ratio := float(actual.get(id,0.0))/float(expected[id])
		minimum = minf(minimum,ratio)
		maximum = maxf(maximum,ratio)
		total += float(expected[id])
	return {"expected_length":total,"expected_by_road":expected,"actual_by_road":actual,"minimum_road_length_ratio":minimum,"maximum_road_length_ratio":maximum,"sample_count":samples,"covered_samples":covered,"sample_coverage":float(covered)/float(samples)}
