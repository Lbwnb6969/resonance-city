extends SceneTree
const World = preload("res://scripts/world/continuous_world.gd")
const Charter = preload("res://scripts/world/development_charter.gd")
const Planning = preload("res://scripts/world/city_spatial_planning.gd")
const Model = preload("res://scripts/audio/continuous_audio_model.gd")
const Renderer = preload("res://scripts/audio/sample_renderer.gd")
const Job = preload("res://scripts/world/world_settlement_job.gd")
var checks: Dictionary = {}
var comparisons: Array = []
const OUT = "res://验证记录/连续河谷/第141轮_规划玩法"
func _initialize() -> void: call_deferred("_run")
func check(ok: bool,label: String) -> void:
	checks[label] = ok
	if not ok: push_error(label)
func pick(world: ContinuousWorld,cid: String = "") -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for lot in world.lots:
		if not str(lot["occupant"]).is_empty() or (not cid.is_empty() and str(lot["community_id"]) != cid): continue
		var p := Vector2(float(lot["center"][0]),float(lot["center"][1]))
		var candidate := world.planning_candidate(p)
		if candidate.is_empty() or not bool(candidate["allowed"]) or candidate.has("public_id"): continue
		if not Planning.public_access(world,candidate).is_empty(): result.append(candidate)
	return result
func step(world: ContinuousWorld) -> void: world.settle((world.phrase_index+1)*int(world.cfg["timing"]["phrase_frames"]))
func _run() -> void:
	var base := World.new()
	base.build_starter("bright")
	var sites := pick(base)
	check(sites.size() >= 2,"two real vacant anchor sites")
	var selected: Dictionary = sites[0]
	var scope: Array = [selected["community_id"]]
	var frame: int = base.cfg["timing"]["phrase_frames"]
	var untouched := base.snapshot()
	check(not base.request({"kind":"charter","plan_id":"missing","community_ids":scope},frame)["accepted"] and base.snapshot() == untouched,"invalid charter leaves world and RNG untouched")
	check(not base.request({"kind":"charter","plan_id":"compact","community_ids":["missing"]},frame)["accepted"] and base.snapshot() == untouched,"invalid scope leaves world untouched")
	var queued: ContinuousWorld = base.fork()
	check(queued.request({"kind":"charter","plan_id":"compact","community_ids":scope+scope},frame)["accepted"],"charter queued at phrase boundary")
	check(queued.development_charters.is_empty() and queued.pending_actions[0]["community_ids"] == scope,"scope canonical; no premature charter")
	queued.request({"kind":"civic","position":selected["position"]},frame)
	var preview := Charter.anchor_preview(queued,selected,"civic",Planning.public_access(queued,selected))
	var job := Job.new()
	job.start(queued,[1],frame)
	job.join()
	job.adopt(queued)
	check(queued.development_charters.size() == 1,"worker adopts charter")
	var anchor := queued.public_at(Vector2(float(selected["position"][0]),float(selected["position"][1])))
	check(not anchor.is_empty() and anchor["anchor_id"] == anchor["id"] and anchor["created_phrase"] == 1,"anchor identity and footprint committed")
	var old_ids: Array = base.by_id.keys()
	var initial_covered := 0
	for id in anchor.get("covered",[]):
		if old_ids.has(id): initial_covered += 1
	check(initial_covered == int(preview["reachable"]),"anchor preview uses actual entrance coverage")
	check(anchor["service_capacity"] == 18 and anchor["polygon"] == selected["polygon"],"real footprint and civic capacity")
	var restore: ContinuousWorld = base.fork()
	restore.restore(queued.snapshot())
	check(restore.development_charters == queued.development_charters and restore.public_sites == queued.public_sites,"snapshot retains charter and canonical anchors")
	var json_saved: Dictionary = JSON.parse_string(JSON.stringify(queued.snapshot()))
	restore.restore(json_saved)
	check(restore.development_charters == queued.development_charters and restore.public_at(Vector2(float(selected["position"][0]),float(selected["position"][1])))["polygon"] == selected["polygon"],"JSON DNA geometry and charter round trip")
	var legacy := base.snapshot()
	legacy.erase("development_charters")
	restore.restore(legacy)
	check(restore.development_charters.is_empty(),"legacy city defaults to no charter")
	queued.communities["test-child"] = {"id":"test-child","parent_id":str(scope[0]),"members":[]}
	check(Charter.for_community(queued,"test-child")["plan_id"] == "compact","split community inherits parent charter")
	queued.request({"kind":"charter","plan_id":"corridor","community_ids":["test-child"]},frame*2)
	Charter.commit(queued,queued.pending_actions.back(),frame*2)
	check(Charter.for_community(queued,"test-child")["plan_id"] == "corridor" and Charter.for_community(queued,str(scope[0]))["plan_id"] == "compact","local charter overrides inherited choice")
	var cancel_model := Model.new(base.fork())
	cancel_model.submit({"kind":"charter","plan_id":"compact","community_ids":scope})
	cancel_model.cancel_pending()
	check(cancel_model.world.pending_actions.is_empty() and cancel_model.world.development_charters.is_empty() and cancel_model.world.buildings == base.buildings and cancel_model.world.rng.snapshot() == base.rng.snapshot(),"cancel removes intent without changing true city")
	var held_rng := queued.rng.snapshot()
	Charter.influence_edges(queued,str(scope[0]),Vector2(float(selected["position"][0]),float(selected["position"][1])))
	check(queued.rng.snapshot() == held_rng,"derived influence consumes no formal RNG")
	for plan: String in ["compact","corridor","bluegreen"]:
		await compare(base,plan,selected,"same-anchor-"+plan)
	await compare(base,"bluegreen",sites[sites.size()-1],"different-anchor-bluegreen")
	check(comparisons[0]["positions"] != comparisons[1]["positions"],"same anchor different charter changes actual growth")
	check(comparisons[2]["positions"] != comparisons[3]["positions"],"same charter different anchor changes actual growth")
	await gpu_flow()
	var ok: bool = not checks.values().has(false)
	var file := FileAccess.open(OUT+".json",FileAccess.WRITE)
	file.store_string(JSON.stringify({"status":"PASS" if ok else "FAIL","checks":checks,"comparisons":comparisons,"scope":"First charter/scope/public anchor slice; cards and human listening pending"}, "\t"))
	print("PLANNING_GAMEPLAY ", "PASS" if ok else "FAIL"," checks=",checks.size())
	quit(0 if ok else 1)
func compare(base: ContinuousWorld,plan: String,site: Dictionary,label: String) -> void:
	var world: ContinuousWorld = base.fork()
	var frame: int = world.cfg["timing"]["phrase_frames"]
	var scope: Array = world.communities.keys()
	world.request({"kind":"charter","plan_id":plan,"community_ids":scope},frame)
	world.request({"kind":"park","position":site["position"]},frame)
	for i in range(8): step(world)
	var positions: Array = world.buildings.map(func(b: Dictionary) -> Array: return b["position"])
	var anchor := world.public_at(Vector2(float(site["position"][0]),float(site["position"][1])))
	var intrusion := false
	for building in world.buildings:
		if Planning.disc_hits(site["polygon"],Vector2(float(building["position"][0]),float(building["position"][1])),float(building["radius"])): intrusion = true
	check(not intrusion,label+" growth respects public footprint")
	check(world.buildings.size() > base.buildings.size(),label+" ordinary growth continues")
	var model := Model.new(world)
	var state: Dictionary = {}
	var pcm := PackedVector2Array()
	pcm.resize(48000*30)
	var start: int = model.next_boundary-int(world.cfg["timing"]["phrase_frames"])
	for offset in range(0,pcm.size(),1024):
		var block := model.render_block(state,start+offset,mini(1024,pcm.size()-offset),48000)
		for i in range(block.size()): pcm[offset+i] = block[i]
	var wav := AudioStreamWAV.new()
	wav.format = AudioStreamWAV.FORMAT_16_BITS
	wav.stereo = true
	wav.mix_rate = 48000
	wav.data = Renderer.pcm16(pcm)
	wav.save_to_wav(OUT+"_"+label+".wav")
	comparisons.append({"label":label,"buildings":world.buildings.size(),"positions":positions,
		"anchor_id":anchor.get("anchor_id",""),"covered":anchor.get("covered",[]),"service":world.public_service_summary(),
		"roles":model.synth.score_snapshot(),"audio_seconds":30,"world_version":Charter.version(world)})
	await process_frame
func gpu_flow() -> void:
	var demo: ContinuousValleyDemo = load("res://scenes/continuous_valley.tscn").instantiate()
	demo.session_persistence = false
	root.add_child(demo)
	await RenderingServer.frame_post_draw
	demo.clock.stop_output()
	demo.set_process(false)
	check(demo.planning_guide.visible and demo.planning_guide.stage == "charter","new city shows three charter cards")
	root.get_texture().get_image().save_png(OUT+"_三章程.png")
	demo.planning_guide.choose_plan("bluegreen")
	var candidates := pick(demo.model.world)
	var site: Dictionary = candidates[0]
	var p := Vector2(float(site["position"][0]),float(site["position"][1]))
	check(demo.planning_guide.map_click(p,false) and demo.planning_guide.scope.has(site["community_id"]),"map selects real stable community")
	demo.planning_guide.commit_scope()
	demo.pointer = demo.point(p)
	demo.planning_guide.sync()
	demo.queue_redraw()
	await RenderingServer.frame_post_draw
	check(demo.planning_guide.stage == "anchor" and not demo.planning_guide.candidate.is_empty(),"anchor footprint influence and capacity preview")
	root.get_texture().get_image().save_png(OUT+"_选区与锚点.png")
	demo.planning_guide.choose_anchor("civic")
	check(demo.planning_guide.map_click(p,false) and not demo.planning_guide.opened,"map places selected facility and collapses guide")
	check(demo.model.world.pending_actions.size() == 2,"charter and anchor queued together")
	var before := demo.model.world.snapshot()
	var frozen := demo.model.freeze()
	var pcm := frozen.forecast({"kind":"civic","position":site["position"]},{},0,4800)
	check(not pcm.is_empty() and demo.model.world.snapshot() == before,"real planning forecast preserves live city and RNG")
	var job := Job.new()
	job.start(demo.model.world,[1],demo.model.next_boundary)
	job.join()
	job.adopt(demo.model.world)
	demo.shown_world = demo.model.world
	demo.planning_guide.sync()
	demo._sync_buildings()
	demo.queue_redraw()
	await RenderingServer.frame_post_draw
	check(demo.shown_world.development_charters.size() == 1 and demo.shown_world.public_at(p).get("anchor_id","") != "","GPU scene reveals committed planning")
	root.get_texture().get_image().save_png(OUT+"_落成.png")
	demo.planning_guide.open_tool()
	check(demo.planning_guide.opened,"P tool can reopen for another community")
	demo.planning_guide.hide_tool()
	demo.free()
	await process_frame
