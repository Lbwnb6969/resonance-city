extends SceneTree
const Scene = preload("res://scenes/continuous_valley.tscn")
const Ground = preload("res://scripts/visual/valley_ground_layer.gd")
const Job = preload("res://scripts/world/world_settlement_job.gd")
var checks: Dictionary = {}
var samples: Array = []
var max_pixel_difference := 0
static func reference_script() -> GDScript:
	var script := GDScript.new()
	script.source_code = FileAccess.get_file_as_string("res://废弃/scripts/2026-10-10_第141轮城市地面保留前/valley_ground_layer.gd.txt").replace("\r","")
	assert(script.reload() == OK)
	return script
func _initialize() -> void: call_deferred("_run")
func _counts(actual: Node2D) -> Dictionary:
	var counts: Dictionary = {}
	for name in actual.urban_parts: counts[name] = actual.urban_parts[name].rebuild_count
	return counts
func _compare(demo: ContinuousValleyDemo,actual: Node2D,old: Node2D,label: String,only: String = "") -> void:
	demo._prepare_ground()
	var before := _counts(actual)
	actual.sync(demo)
	old.sync(demo)
	actual.show()
	old.hide()
	await RenderingServer.frame_post_draw
	await RenderingServer.frame_post_draw
	var a := actual.get_viewport().get_texture().get_image()
	actual.hide()
	old.show()
	await RenderingServer.frame_post_draw
	await RenderingServer.frame_post_draw
	var b := actual.get_viewport().get_texture().get_image()
	var aa := a.get_data()
	var bb := b.get_data()
	var difference := 0
	if aa != bb:
		for index in range(aa.size()): difference = maxi(difference,absi(int(aa[index])-int(bb[index])))
		a.save_png("res://验证记录/连续河谷/第141轮_不等价_"+label+"_新.png")
		b.save_png("res://验证记录/连续河谷/第141轮_不等价_"+label+"_旧.png")
	max_pixel_difference = maxi(max_pixel_difference,difference)
	checks[label+" exact pixels"] = aa == bb
	var deltas: Dictionary = {}
	for name in actual.urban_parts: deltas[name] = actual.urban_parts[name].rebuild_count-int(before.get(name,0))
	if not only.is_empty():
		checks[label+" affected parts"] = deltas.values().all(func(value: int) -> bool: return value == 0) if only == "none" else deltas.keys().all(func(name: String) -> bool: return int(deltas[name]) == (1 if name == only else 0))
	samples.append({"label":label,"draw_ms":actual.last_draw_ms,"prepare_ms":actual.last_prepare_ms,"old_draw_ms":old.last_draw_ms,"part_draws":deltas,"byte_difference":difference})
	old.hide()
	actual.hide()
func _run() -> void:
	var demo: ContinuousValleyDemo = Scene.instantiate()
	demo.session_persistence = false
	root.add_child(demo)
	demo.clock.stop_output()
	demo.set_process(false)
	demo.model.close_background_settlement()
	demo.set_anchors_preset(Control.PRESET_TOP_LEFT)
	demo.size = Vector2(1024,768)
	var viewport := SubViewport.new()
	viewport.size = Vector2i(1024,768)
	viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(viewport)
	var background := ColorRect.new()
	background.color = Color(0.7,0.7,0.7)
	background.size = Vector2(viewport.size)
	viewport.add_child(background)
	var actual := Ground.new()
	var old: Node2D = reference_script().new()
	actual.role = "urban"
	old.role = "urban"
	viewport.add_child(actual)
	viewport.add_child(old)
	for family in ValleyTerrain.family_ids():
		var city := ContinuousWorld.new("2026100900000062",true,family)
		city.build_starter("bright")
		demo.model = ContinuousAudioModel.new(city)
		demo.shown_world = city
		demo.view_zoom = 0.88
		demo.pan = Vector2.ZERO
		await _compare(demo,actual,old,family+" opening")
		city.settle(int(city.cfg["timing"]["phrase_frames"]))
		await _compare(demo,actual,old,family+" growth")
		var counts := _counts(actual)
		demo.pan += Vector2(30,-50)
		await _compare(demo,actual,old,family+" pan","none")
		checks[family+" camera shares local transforms"] = actual.position == old.position and actual.scale == old.scale and _counts(actual) == counts
		demo.view_zoom = 0.52
		await _compare(demo,actual,old,family+" zoom","public")
	var city := ContinuousWorld.regional_sample()
	demo.model = ContinuousAudioModel.new(city)
	demo.shown_world = city
	demo.view_zoom = 0.20
	demo.pan = Vector2(220,-80)
	var world_before := city.snapshot()
	await _compare(demo,actual,old,"regional1511")
	checks["draw leaves world unchanged"] = city.snapshot() == world_before
	var job := Job.new()
	job.start(city,[],int(city.cfg["timing"]["phrase_frames"])*(city.phrase_index+1))
	while not job.completed(): await process_frame
	job.join()
	job.adopt(city)
	await _compare(demo,actual,old,"regional1525 growth","none")
	checks["genuine natural growth"] = city.buildings.size() == 1525
	# Exercise each content signature, including direct mutation of original arrays.
	city.centers[0]["position"][0] += 0.2
	city.phrase_index += 1
	await _compare(demo,actual,old,"center moved","centers")
	city.green_corridors[0]["path"][0] += Vector2(0.3,0.1)
	city.phrase_index += 1
	await _compare(demo,actual,old,"corridor original geometry changed","corridors")
	var site: Dictionary = city.public_sites[0]
	site["active"] = not site["active"]
	city.phrase_index += 1
	await _compare(demo,actual,old,"public activation","public")
	var civic: Dictionary = {}
	for item in city.public_sites:
		if item["kind"] == "civic" and item["active"]:
			civic = item
			break
	assert(not civic.is_empty())
	civic["service_level"] = int(civic.get("service_level",1))+1
	civic["served"] = []
	city.phrase_index += 1
	await _compare(demo,actual,old,"civic level and saturation","public")
	city.planning_reservations.append({"polygon":city.lots[0]["polygon"],"id":"test-reservation"})
	city.phrase_index += 1
	await _compare(demo,actual,old,"reservation","public")
	city.phrase_index += 1
	demo._prepare_ground()
	demo.road_draw_polygons[0]["color"] = Color("aabb99")
	await _compare(demo,actual,old,"road color","roads")
	for lot in city.lots:
		if lot["occupant"] != "":
			lot["polygon"][0] += Vector2(0.2,0.1)
			break
	city.phrase_index += 1
	await _compare(demo,actual,old,"occupied lot original polygon changed","lots")
	var moved: String = city.fabric_cache["shared_walls"][0]["a"]
	city.by_id[moved]["position"][0] += 0.25
	city.phrase_index += 1
	await _compare(demo,actual,old,"fabric building moved","fabric")
	city.fabric_cache = {}
	city.phrase_index += 1
	await _compare(demo,actual,old,"fabric removed","fabric")
	checks["removed geometry cache pruned"] = actual.urban_parts["fabric"].projected.is_empty() and actual.urban_parts["fabric"].point_cache.is_empty()
	city.terrain.river_phase += 0.13
	city.phrase_index += 1
	await _compare(demo,actual,old,"terrain changed")
	var terrain_count: int = actual.urban_parts["roads"].rebuild_count
	checks["terrain invalidates retained parts"] = samples[-1]["part_draws"].values().all(func(value: int) -> bool: return value == 1)
	demo.black = true
	actual.sync(demo)
	checks["black listening hides every part"] = not actual.visible and actual.urban_parts.values().all(func(part: Node2D) -> bool: return not part.is_visible_in_tree())
	demo.black = false
	await _compare(demo,actual,old,"black restored","none")
	checks["black restore retains commands"] = actual.urban_parts["roads"].rebuild_count == terrain_count
	var passed := checks.values().all(func(ok: bool) -> bool: return ok)
	var result := {"passed":passed,"checks":checks,"samples":samples,"max_pixel_difference":max_pixel_difference,
		"scope":"GPU exact RGBA comparison with archived single CanvasItem: five terrain families, natural 1511→1525 growth, six independent parts, mutations/removal/camera and black view",
		"source_sha256":FileAccess.get_sha256("res://scripts/visual/valley_ground_layer.gd")}
	FileAccess.open("res://验证记录/连续河谷/第141轮_地面保留合同.json",FileAccess.WRITE).store_string(JSON.stringify(result,"\t")+"\n")
	print("GROUND_RETENTION ",JSON.stringify(result))
	viewport.queue_free()
	demo.queue_free()
	await process_frame
	quit(0 if passed else 1)
