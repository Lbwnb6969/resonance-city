class_name CityCreationGrassMask
extends RefCounted
## Dirty-block-only spatial index. Each bin delegates to the authoritative
## vegetation rule; houses retain all cells so door exposure stays exact.
const State := preload("res://scripts/world/city_creation_state.gd")
const BLOCK_SIZE := 8.0
const BIN_SIZE := 1.0
const BIN_COUNT := 8
const BIN_PADDING := 1.0
const MAX_CLEARANCE := 0.42 * 0.25

var _origin := Vector2.ZERO
var _bounds := Rect2()
var _bins: Array[RefCounted] = []
var _candidate := {"id": "pgrassmask", "position": [0.0, 0.0], "kind": "shrub", "scale": 0.18, "seed": 0, "source": "seed"}
var _stats: Dictionary = {}

func setup(state: RefCounted, block: Vector2i) -> void:
	var started := Time.get_ticks_usec()
	_origin = Vector2(block) * BLOCK_SIZE
	_bounds = Rect2(_origin, Vector2.ONE * BLOCK_SIZE)
	_bins.clear()
	for index in range(BIN_COUNT * BIN_COUNT):
		_bins.append(State.new())
	for house: Dictionary in state.get("houses"):
		var touched: Dictionary = {}
		var origin := Vector2(float(house.origin[0]), float(house.origin[1]))
		for cell: Dictionary in house.cells:
			if int(cell.level) != 0:
				continue
			var at := origin + Vector2(float(cell.x), float(cell.z)) * State.CELL_UNIT
			var footprint := Rect2(at - Vector2.ONE * State.CELL_UNIT * 0.5, Vector2.ONE * State.CELL_UNIT).grow(MAX_CLEARANCE)
			for index: int in _indices(footprint):
				touched[index] = true
			if cell.kind == "room":
				var radius := 0.95 + MAX_CLEARANCE
				var doorway := Rect2(at + Vector2(0, 1.55) - Vector2.ONE * radius, Vector2.ONE * radius * 2.0)
				for index: int in _indices(doorway):
					touched[index] = true
		for index: int in touched:
			_bins[index].houses.append(house)
	for road: Dictionary in state.get("roads"):
		for index in range(1, road.points.size()):
			var a := Vector2(float(road.points[index - 1][0]), float(road.points[index - 1][1]))
			var b := Vector2(float(road.points[index][0]), float(road.points[index][1]))
			var segment := {"points": [[a.x, a.y], [b.x, b.y]], "width": road.width}
			for bin_index: int in _indices(_segment_bounds(a, b, float(road.width) * 0.5 + 0.45 + MAX_CLEARANCE)):
				_bins[bin_index].roads.append(segment)
	for water: Dictionary in state.get("waters"):
		for index in range(maxi(1, water.points.size() - 1)):
			var a := Vector2(float(water.points[index][0]), float(water.points[index][1]))
			var next := mini(index + 1, water.points.size() - 1)
			var b := Vector2(float(water.points[next][0]), float(water.points[next][1]))
			var segment := {"points": [[a.x, a.y], [b.x, b.y]], "width": water.width}
			for bin_index: int in _indices(_segment_bounds(a, b, float(water.width) * 0.5 + MAX_CLEARANCE)):
				_bins[bin_index].waters.append(segment)
	var houses := 0
	var roads := 0
	var waters := 0
	var max_segments := 0
	for bin: RefCounted in _bins:
		houses += bin.houses.size()
		roads += bin.roads.size()
		waters += bin.waters.size()
		max_segments = maxi(max_segments, bin.roads.size() + bin.waters.size())
	_stats = {"bins": _bins.size(), "house_references": houses, "road_segments": roads,
		"water_segments": waters, "max_bin_segments": max_segments,
		"setup_ms": float(Time.get_ticks_usec() - started) / 1000.0}

func visible(at: Vector2, scale: float) -> bool:
	if not _bounds.has_point(at) or not State.WORLD_RECT.has_point(at):
		return false
	var local := at - _origin
	var index := floori(local.y / BIN_SIZE) * BIN_COUNT + floori(local.x / BIN_SIZE)
	_candidate.position[0] = at.x
	_candidate.position[1] = at.y
	_candidate.scale = scale
	return _bins[index].vegetation_visible(_candidate)

func stats() -> Dictionary:
	return _stats.duplicate()

func _segment_bounds(a: Vector2, b: Vector2, radius: float) -> Rect2:
	return Rect2(Vector2(minf(a.x, b.x), minf(a.y, b.y)), Vector2(absf(a.x - b.x), absf(a.y - b.y))).grow(radius)

func _indices(bounds: Rect2) -> Array[int]:
	var result: Array[int] = []
	# Padding is extra broad-phase coverage only. Final visibility always uses
	# the exact foundation, door, capsule and base-river predicates.
	var expanded := bounds.grow(BIN_PADDING)
	if not expanded.intersects(_bounds):
		return result
	var first := Vector2i(floori((expanded.position.x - _origin.x) / BIN_SIZE), floori((expanded.position.y - _origin.y) / BIN_SIZE))
	var last := Vector2i(floori((expanded.end.x - _origin.x) / BIN_SIZE), floori((expanded.end.y - _origin.y) / BIN_SIZE))
	for z in range(maxi(0, first.y), mini(BIN_COUNT - 1, last.y) + 1):
		for x in range(maxi(0, first.x), mini(BIN_COUNT - 1, last.x) + 1):
			result.append(z * BIN_COUNT + x)
	return result
